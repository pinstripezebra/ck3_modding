# Magical pilgrimage activity — replaces the old decision, provides the real
# CK3 map location picker with 8 hard-coded arcane sites across the world.

activity_wizard_magic_pilgrimage = {
	is_shown = {
		always = no
		is_playable_character = yes
		is_adult = yes
		OR = {
			wizard_has_any_lore_trait = yes
			wizard_has_any_magic_potential_trait = yes
		}
		trigger_if = {
			limit = { is_ai = yes }
			is_at_war = no
			gold >= 600
		}
	}

	can_start_showing_failures_only = {
		NOT = { is_activity_type_on_cooldown = activity_wizard_magic_pilgrimage }
		is_available_adult = yes
		custom_tooltip = {
			text = wizard_magic_pilgrimage_gold_requirement
			gold >= 500
		}
	}

	is_valid = {
		scope:host = {
			is_alive = yes
			is_imprisoned = no
			NOT = { is_incapable = yes }
		}
	}

	# Only the 8 arcane sites are valid destinations.
	is_location_valid = {
		barony ?= {
			OR = {
				this = title:b_jerusalem
				this = title:b_alexandria
				this = title:b_constantinople
				this = title:b_baghdad
				this = title:b_samarkand
				this = title:b_toledo
				this = title:b_roma
				this = title:b_athens
			}
		}
	}

	province_description = {
		desc = wizard_magic_pilgrimage_province_desc
	}

	on_invalidated = {
		scope:host = {
			remove_character_modifier = wizard_magic_pilgrimage_traveling
		}
	}

	# Deduct gold when the activity is confirmed.
	on_start = {
		scope:host = {
			add_gold = -500
		}
	}

	# When the host arrives at the arcane site, fire the location-specific
	# flavor event, which chains to the lore-granting event after 120 days.
	on_enter_active_state = {
		if = {
			limit = { this = scope:host }
			add_character_modifier = {
				modifier = wizard_magic_pilgrimage_traveling
				months = 4
			}
			if = {
				limit = { scope:activity.activity_location.barony ?= title:b_jerusalem }
				trigger_event = wizard_magic_pilgrimage.1001
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_alexandria }
				trigger_event = wizard_magic_pilgrimage.1002
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_constantinople }
				trigger_event = wizard_magic_pilgrimage.1003
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_baghdad }
				trigger_event = wizard_magic_pilgrimage.1004
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_samarkand }
				trigger_event = wizard_magic_pilgrimage.1005
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_toledo }
				trigger_event = wizard_magic_pilgrimage.1006
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_roma }
				trigger_event = wizard_magic_pilgrimage.1007
			}
			else_if = {
				limit = { scope:activity.activity_location.barony ?= title:b_athens }
				trigger_event = wizard_magic_pilgrimage.1008
			}
			else = {
				trigger_event = wizard_magic_pilgrimage.1001
			}
		}
	}
}
