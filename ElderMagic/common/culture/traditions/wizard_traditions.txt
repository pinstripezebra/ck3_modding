# Elder Magic — Cultural Traditions

tradition_sorcerous_heritage = {
	category = societal

	layers = {
		0 = learning
		1 = western
		4 = sorcerous_heritage.dds
	}

	parameters = {
		sorcerous_heritage_lore_opinion = yes
	}

	# CK3 only registers MaA modifiers per archetype, not per unit key; every
	# Elder Magic mage regiment uses the custom magi type.
	character_modifier = {
		magi_max_size_add = 4
		magi_recruitment_cost_mult = -0.2
		magi_maintenance_mult = -0.2
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_spiritual = { is_in_list = traits }
						culture_pillar:ethos_courtly = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_spiritual_or_courtly_desc
				}
			}
			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_sorcerous_heritage }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = { value = 50 }
}
