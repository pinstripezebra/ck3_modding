# Ascendant Lore Decisions — one per Lore of Magic.
# Visible when: character has the lore trait at max level, does not yet have
# the ascendant trait, and is not on the 5-year failure cooldown.
# Fires the corresponding wizard_ascendant.100X ritual event which resolves
# into success (sub-event 200X) or failure (sub-event 200Y).

wizard_ascendant_fire_decision = {
	title = wizard_ascendant_fire_decision
	desc = wizard_ascendant_fire_decision_desc
	selection_tooltip = wizard_ascendant_fire_decision_tooltip
	confirm_text = wizard_ascendant_decision_confirm
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_ascendant_fire.dds"
	}
	ai_check_interval = 24

	is_shown = {
		is_adult = yes
		has_trait = lore_of_fire
		OR = {
			has_character_flag = wizard_lore_fire_maxed
			AND = {
				has_variable = wizard_fire_xp_total
				var:wizard_fire_xp_total >= 100
			}
		}
		NOT = { has_trait = ascendant_lore_of_fire }
		NOT = { has_character_modifier = wizard_ascendant_fire_failed }
	}

	is_valid = {
		is_adult = yes
		is_imprisoned = no
		custom_description = {
			text = wizard_lore_fire_mastery_requirement
			OR = {
				has_character_flag = wizard_lore_fire_maxed
				AND = {
					has_variable = wizard_fire_xp_total
					var:wizard_fire_xp_total >= 100
				}
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 30
	}

	effect = {
		trigger_event = { id = wizard_ascendant.1001 }
	}
}
