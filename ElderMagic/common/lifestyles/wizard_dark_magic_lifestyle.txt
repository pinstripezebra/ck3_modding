
# Dark Magic — the wizard lifestyle. Restricted to necromancers: only valid
# for characters who have awakened the Lore of Death trait.
dark_magic_lifestyle = {
	# is_valid/is_highlighted can't use scripted triggers (see vanilla
	# common/lifestyles/_lifestyles.info).
	is_valid = {
		has_trait = lore_of_death
	}

	is_highlighted = {
		has_trait = lore_of_death
	}

	xp_per_level = 1000
	base_xp_gain = 25
}
