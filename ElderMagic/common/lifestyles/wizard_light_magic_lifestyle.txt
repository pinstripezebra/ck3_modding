
# Light Magic — the wizard lifestyle. Valid for elemental/high-magic lore
# traits; Lore of Death belongs to Dark Magic instead.
light_magic_lifestyle = {
	# is_valid/is_highlighted can't use scripted triggers (see vanilla
	# common/lifestyles/_lifestyles.info).
	is_valid = {
		OR = {
			has_trait = lore_of_fire
			has_trait = lore_of_earth
			has_trait = lore_of_water
			has_trait = lore_of_high_magic
		}
	}

	is_highlighted = {
		OR = {
			has_trait = lore_of_fire
			has_trait = lore_of_earth
			has_trait = lore_of_water
			has_trait = lore_of_high_magic
		}
	}

	xp_per_level = 1000
	base_xp_gain = 25
}
