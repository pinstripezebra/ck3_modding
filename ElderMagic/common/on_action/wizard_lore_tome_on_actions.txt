# 1% yearly chance for any playable character to stumble upon a ruin rumored to
# hold an ancient tome of magical lore. Mirrors wizard_ruins_on_actions.txt's
# pattern with its own separate cooldown flag.

random_yearly_playable_pulse = {
	on_actions = { wizard_lore_tome_travel_check }
}

wizard_lore_tome_travel_check = {
	effect = {
		if = {
			limit = {
				is_adult = yes
				is_imprisoned = no
				NOT = { has_character_flag = wizard_lore_tome_cooldown }
			}
			random = {
				chance = 1
				trigger_event = wizard_lore_tome.0001
				add_character_flag = { flag = wizard_lore_tome_cooldown days = 1825 }
			}
		}
	}
}
