# Confirmed working pattern from CK3 wiki (Shrek TC example):
# on_game_start + on_actions = {} is append-safe and fires in global scope.
# set_global_variable works from global scope - no every_player needed.
on_game_start = {
	on_actions = { wizard_mana_init }
}

wizard_mana_init = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = wizard_mana_start_year } }
			set_global_variable = { name = wizard_mana_start_year value = current_year }
		}
	}
}
