# on_game_start fires in global scope; set_global_variable works from there.
on_game_start = {
	on_actions = { wizard_mana_init }
}

wizard_mana_init = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = wizard_mana_start_year } }
			set_global_variable = { name = wizard_mana_start_year value = current_year }
		}
	}
}

# quarterly_playable_pulse is the most frequent scripted pulse in CK3 (fires 4x per year)
quarterly_playable_pulse = {
	events = { wizard_mana.0002 }
}
