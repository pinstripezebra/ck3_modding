# Elemental Magic and Dark Magic crime doctrines, mirroring vanilla's doctrine_witchcraft_*
# (common/religion/doctrine_types/20_doctrines.txt) piety_cost/parameters structure.

doctrine_elemental_magic_crime = {
	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { has_doctrine = doctrine_elemental_magic_crime }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { has_doctrine = doctrine_elemental_magic_accepted }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { has_doctrine = doctrine_elemental_magic_virtuous }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
	}

	parameters = {
		elemental_magic_illegal = yes
	}
}
