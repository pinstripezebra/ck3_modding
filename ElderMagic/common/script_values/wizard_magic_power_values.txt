# Total Magic Power — computed on demand, no stored state required.
# +1 per XP tier reached in each lore trait (max 4 per trait, 20 total).
# +rank for wizard potential ladder (the_spark=1 ... the_horizon=10).

# Per-lore contributions (0–4 each), used by the magic power breakdown tooltip.
wizard_fire_lore_power = {
	value = 0
	if = { limit = { has_trait_xp = { trait = lore_of_fire value >= 25  } } add = 1 }
	if = { limit = { has_trait_xp = { trait = lore_of_fire value >= 50  } } add = 1 }
	if = { limit = { has_trait_xp = { trait = lore_of_fire value >= 75  } } add = 1 }

}
