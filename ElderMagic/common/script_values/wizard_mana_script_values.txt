# Computes the monthly mana generation rate for the current character.
# Read live from the GUI — no stored variable, no events, always accurate.
wizard_mana_gain_rate = {
	value = 0
	# Wizard potential ladder (mutually exclusive, +1 to +20)
	if = { limit = { has_trait = the_spark }       add = 1 }
	if = { limit = { has_trait = the_conduit }     add = 2 }
	if = { limit = { has_trait = the_resonator }   add = 4 }
	if = { limit = { has_trait = the_vector }      add = 6 }
	if = { limit = { has_trait = the_capacitor }   add = 8 }
	if = { limit = { has_trait = the_dynamo }      add = 10 }
	if = { limit = { has_trait = the_nexus }       add = 12 }
	if = { limit = { has_trait = the_singularity } add = 14 }
	if = { limit = { has_trait = the_catalyst }    add = 16 }
	if = { limit = { has_trait = the_horizon }     add = 20 }
	# Magic education (stackable with ladder, +1 to +5)
	if = { limit = { has_trait = magic_education_1 } add = 1 }
	if = { limit = { has_trait = magic_education_2 } add = 2 }
	if = { limit = { has_trait = magic_education_3 } add = 3 }
	if = { limit = { has_trait = magic_education_4 } add = 4 }
	if = { limit = { has_trait = magic_education_5 } add = 5 }

	# Light Magic lifestyle perks (Battlemage tree)
	if = { limit = { has_perk = overcharged_casting_perk } add = 2 }
	if = { limit = { has_perk = arcane_reserves_perk     } add = 3 }
}

# Multiplier applied to spell mana costs — stacks the Battlemage tree's two
# casting-efficiency perks. Floored at 0.5 so costs can never hit zero.
wizard_spell_cost_discount_mult = {
	value = 1
	if = { limit = { has_perk = efficient_casting_perk } subtract = 0.2 }
	if = { limit = { has_perk = arcane_reserves_perk   } subtract = 0.15 }
	min = 0.5
}
wizard_spell_mana_cost_tier_1 = { value = 100 multiply = wizard_spell_cost_discount_mult }
wizard_spell_mana_cost_tier_2 = { value = 220 multiply = wizard_spell_cost_discount_mult }
wizard_spell_mana_cost_tier_3 = { value = 400 multiply = wizard_spell_cost_discount_mult }
wizard_spell_mana_cost_tier_4 = { value = 600 multiply = wizard_spell_cost_discount_mult }

wizard_mana_annual_rate = {
	value = 0
	add = wizard_mana_gain_rate
	multiply = 12
}

# wizard_mana (gross accumulated) grows monthly via wizard_mana.0002 event.
# Falls back to yearly formula for saves before the first monthly tick.
wizard_mana_total = {
	value = 0
	# Primary: stored monthly gross pool
	if = {
		limit = { has_global_variable = wizard_mana }
		add = global_var:wizard_mana
	}
	if = {
		limit = { has_global_variable = wizard_mana  has_global_variable = wizard_mana_spent }
		subtract = global_var:wizard_mana_spent
	}
	# Legacy fallback: yearly formula until first monthly event fires
	if = {
		limit = { NOT = { has_global_variable = wizard_mana }  has_global_variable = wizard_mana_start_year }
		add = current_year
		subtract = global_var:wizard_mana_start_year
		multiply = 12
		multiply = wizard_mana_gain_rate
	}
	if = {
		limit = { NOT = { has_global_variable = wizard_mana }  has_global_variable = wizard_mana_start_year  has_global_variable = wizard_mana_spent }
		subtract = global_var:wizard_mana_spent
	}
	min = 0
	max = 1000
}
