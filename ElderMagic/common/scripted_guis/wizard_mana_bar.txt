wizard_mana_bar_visible = {
	is_shown = {
		always = yes
	}
}

# Fires when the mana bar window first shows (game start / save load).
# Sets wizard_mana_start_year in player character scope so wizard_mana_total
# can compute the accumulated total without any periodic event hooks.
wizard_mana_start_init = {
	is_shown = { always = yes }
	effect = {
		if = {
			limit = { NOT = { has_global_variable = wizard_mana_start_year } }
			set_global_variable = { name = wizard_mana_start_year value = current_year }
		}
	}
}

wizard_mana_popup_visible = {
	is_shown = {
		has_global_variable = wizard_mana_popup_open
	}
}

wizard_mana_popup_toggle = {
	is_shown = { always = yes }
	effect = {
		if = {
			limit = { has_global_variable = wizard_mana_popup_open }
			remove_variable = wizard_mana_popup_open
		}
		else = {
			set_variable = { name = wizard_mana_popup_open value = 1 }
		}
	}
}
