# Clears the trigger global var after the GUI bridge widget fires.
# Called from wizard_spellbook_trigger's _show state.
wizard_clear_spellbook_trigger = {
	effect = {
		remove_global_variable = wizard_spellbook_wants_to_open
	}
}

# Clears the close-trigger global var after the close bridge fires.

	effect = {
		remove_global_variable = wizard_spellbook_wants_to_close
	}
}
