wizard_has_any_lore_trait = {
	OR = {
		has_trait = lore_of_fire
		has_trait = lore_of_earth
		has_trait = lore_of_water
		has_trait = lore_of_death
		has_trait = lore_of_high_magic
	}
}

		has_trait = the_spark

		has_trait = the_resonator
		has_trait = the_vector
		has_trait = the_capacitor

		has_trait = the_nexus
		has_trait = the_singularity

}

wizard_has_missing_lore_from_recipient = {
	OR = {
		AND = {
			NOT = { has_trait = lore_of_fire }
			scope:recipient = { has_trait = lore_of_fire }
		}
		AND = {
			NOT = { has_trait = lore_of_earth }
			scope:recipient = { has_trait = lore_of_earth }
		}
		AND = {
			NOT = { has_trait = lore_of_water }
			scope:recipient = { has_trait = lore_of_water }
		}
		AND = {
			NOT = { has_trait = lore_of_death }
			scope:recipient = { has_trait = lore_of_death }
		}
		AND = {
			NOT = { has_trait = lore_of_high_magic }
			scope:recipient = { has_trait = lore_of_high_magic }
		}
	}
}
