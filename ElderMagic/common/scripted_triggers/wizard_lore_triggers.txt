	OR = {
		AND = {
			NOT = { has_trait = lore_of_fire }

		}
		AND = {
			NOT = { has_trait = lore_of_earth }

# Barony-scope checks for the University of Magic Visit destination gate.
# is_location_valid/province_score are evaluated per-province (root = the
# province, confirmed in _activity_type.info) — checking the exact province
# directly (no county/duchy-wide wrap) keeps only the barony that actually
# has the building selectable, not its whole county or duchy.
wizard_barony_has_necromancer_sanctum_trigger = {
	has_building_or_higher = necromancer_sanctum_1
}

wizard_barony_has_elemental_tower_trigger = {
	has_building_or_higher = elemental_tower_1
}
