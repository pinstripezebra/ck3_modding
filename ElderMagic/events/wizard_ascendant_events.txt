﻿namespace = wizard_ascendant

wizard_ascendant.1001 = {
	type = character_event
	title = wizard_ascendant.1001.t
	desc = wizard_ascendant.1001.desc
	theme = learning
	left_portrait = { character = root  animation = personality_bold }
	override_background = { reference = "study" }

	immediate = {
		random = {
			chance = wizard_ascendant_success_chance
			add_character_flag = wizard_ascendant_fire_success
		}
	}

	option = {
		name = wizard_ascendant.ritual_success
		trigger = { has_character_flag = wizard_ascendant_fire_success }
		remove_character_flag = wizard_ascendant_fire_success
		trigger_event = { id = wizard_ascendant.2001 }
	}
	option = {
		name = wizard_ascendant.ritual_failure
		fallback = yes
		trigger = { NOT = { has_character_flag = wizard_ascendant_fire_success } }
		trigger_event = { id = wizard_ascendant.2002 }
	}
}

wizard_ascendant.2002 = {
	type = character_event
	title = wizard_ascendant.2002.t
	desc = wizard_ascendant.2002.desc
	theme = learning
	left_portrait = { character = root  animation = personality_afraid }
	override_background = { reference = "study" }
	immediate = {
		random = {
			chance = 1
			death = { death_reason = death_murder_unknown }
		}
		if = {
			limit = { is_alive = yes }
			random = {
				chance = 20
				add_trait = wounded_2
			}
		}
		if = {
			limit = { is_alive = yes }
			add_character_modifier = { modifier = wizard_ascendant_fire_failed years = 5 }
		}
	}
	option = { name = wizard_ascendant.2002.a }
}
