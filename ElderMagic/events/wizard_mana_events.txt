namespace = wizard_mana

# Fires quarterly for the player wizard; accumulates 3 months of mana into global_var:wizard_mana
wizard_mana.0002 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		wizard_has_any_magic_potential_trait = yes
	}

	immediate = {
		# Migration: seed wizard_mana from formula on first use so existing saves don't lose accumulated mana
		if = {
			limit = { NOT = { has_global_variable = wizard_mana } }
			set_global_variable = { name = wizard_mana value = wizard_mana_total }
			if = {
				limit = { has_global_variable = wizard_mana_spent }
				change_global_variable = { name = wizard_mana add = global_var:wizard_mana_spent }
			}
		}
		# Add 3 months of mana (one quarter)
		change_global_variable = { name = wizard_mana add = wizard_mana_gain_rate }
		change_global_variable = { name = wizard_mana add = wizard_mana_gain_rate }
		change_global_variable = { name = wizard_mana add = wizard_mana_gain_rate }
	}
}
