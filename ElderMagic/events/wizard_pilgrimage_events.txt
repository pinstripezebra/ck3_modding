wizard_magic_pilgrimage.2001 = {
	type = character_event
	title = wizard_magic_pilgrimage.2001.t
	desc = wizard_magic_pilgrimage.2001.desc
	theme = learning
	left_portrait = { character = root animation = personality_content }
	override_background = { reference = "study" }

	immediate = {
		remove_character_modifier = wizard_magic_pilgrimage_traveling

		if = {
			limit = { wizard_missing_any_lore_trait = yes }
			wizard_grant_random_missing_lore = yes
		}
		else = {
			wizard_gain_lore_progress = yes
		}

		remove_character_flag = wizard_magic_destination_1
		remove_character_flag = wizard_magic_destination_2
		remove_character_flag = wizard_magic_destination_3
		remove_character_flag = wizard_magic_destination_4
		remove_character_flag = wizard_magic_destination_5
		remove_character_flag = wizard_magic_destination_6
		remove_character_flag = wizard_magic_destination_7
		remove_character_flag = wizard_magic_destination_8
	}

	option = {
		name = wizard_magic_pilgrimage.option_complete
	}
}

# Hidden intermediary: rolls 50/50 success before showing the result event
wizard_magic_pilgrimage.2000 = {
	type = character_event
	hidden = yes
	is_triggered_only = yes

	immediate = {
		random_list = {
			50 = { trigger_event = wizard_magic_pilgrimage.2001 }
			50 = { trigger_event = wizard_magic_pilgrimage.2002 }
		}
	}
}
