namespace = wizard_spells

wizard_spells.1000 = {
	type = character_event
	hidden = yes
	is_triggered_only = yes

	immediate = {
		wizard_apply_learn_spell_cooldown = yes

		if = {
			limit = { learning >= 25 }
			random = { chance = 90 wizard_learn_selected_spell = yes }
		}
		else_if = {
			limit = { learning >= 20 }
			random = { chance = 78 wizard_learn_selected_spell = yes }
		}
		else_if = {
			limit = { learning >= 15 }
			random = { chance = 62 wizard_learn_selected_spell = yes }
		}
		else_if = {
			limit = { learning >= 10 }
			random = { chance = 45 wizard_learn_selected_spell = yes }
		}
		else = {
			random = { chance = 30 wizard_learn_selected_spell = yes }
		}

		if = {
			limit = { wizard_selected_spell_known = yes }
			trigger_event = wizard_spells.1001
		}
		else = {
			trigger_event = wizard_spells.1002
		}
	}
}

wizard_spells.1001 = {
	type = character_event
	title = wizard_spells.1001.t
	desc = wizard_spells.1001.desc

	option = {
		name = wizard_spells.1001.a
	}
}

	type = character_event
	hidden = yes
	is_triggered_only = yes

	immediate = {

			trigger_event = wizard_spells.2001
